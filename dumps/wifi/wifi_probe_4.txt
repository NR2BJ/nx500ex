===== wlan.sh 내용
#!/bin/sh

MODULE_PATH=/lib/modules
FIRMWARE_PATH=/lib/firmware
MAC_PATH=/opt/etc/.mac.info
REGDB_PATH=/opt/etc/wl-regdom.conf
CAL_DATA_CONF=mrvl/WlanCalData_ext.conf
TX_PWR_PATH=mrvl/txpwrlimit_cfg.bin
NOJAM_PATH=/mnt/mmc/wifi_nojam.txt
# CAMERA_OFF=/opt/etc/nfc.off

# For logging and debugging
ENABLE_LOG=YES
LOG_FILE=/var/log/wlanplug_log

IFACE_NAME=mlan0
NET_IFACE_NAME=mlan0
P2P_IFACE_NAME=p2p0
UAP_IFACE_NAME=uap0

DRV_PARAM="cfg80211_wext=15 txpwrlimit_cfg=${TX_PWR_PATH} cal_data_cfg=${CAL_DATA_CONF}"
DRV_PARAM_NET="drv_mode=1"
DRV_PARAM_MFG="drv_mode=1 mfg_mode=1"
DRV_PARAM_SOFTAP="drv_mode=2 uap_name=uap max_uap_bss=1"
DRV_PARAM_P2P="drv_mode=4 wfd_name=p2p max_vir_bss=1"
DRV_PARAM_CERTI="auto_ds=2 ps_mode=2 hw_test=1"

check_hw()
{
		echo "Hardware Model=NX1-MRVL Revision Number=0.1"

		FIRMWARE_NET=mrvl/sd8887_uapsta.bin
		FIRMWARE_MFT=mrvl/sdio8887_sdio_combo.bin
		FIRMWARE_SOFTAP=mrvl/sd8887_uapsta.bin
		FIRMWARE_P2P=mrvl/sd8887_uapsta.bin
}

regdb()
{
	if [ -e ${REGDB_PATH} ] > /dev/null 2>&1; then
		reg_dom=$(cat ${REGDB_PATH})
		echo "len of mac ${#reg_dom}"
		if [ ${#reg_dom} == 2 ]; then
			echo "reg_dom : ${reg_dom}"
			return
		fi
	fi

	reg_dom=00
	echo "reg_dom : ${reg_dom}"
}

nojam()
{
	if [ -e ${NOJAM_PATH} ] > /dev/null 2>&1; then
		echo "nojam enabled, erase txt to disable"
		rm -rf ${NOJAM_PATH}
	else
		echo "nojam disabled, create txt to enable"
		echo 1 > ${NOJAM_PATH}
	fi
}

mac()
{
	# 1. inspect /opt/etc/.mac.info
	if [ -e ${MAC_PATH} ] > /dev/null 2>&1; then
		_mac=$(cat ${MAC_PATH})
		# echo "len of mac ${#_mac}"
		# echo "mac: ${_mac}"

		if [ ${#_mac} == 17 ]; then
			echo "mac addr exists in .mac.info: ${_mac}"
			return
		fi
	fi

	# 2. generate new MAC addres randomly and store it in /opt/etc/.mac.info
	MAC_PRE=$(echo "a0:21:95:")
	MAC_POST=$(openssl rand -hex 3 | sed 's/\(..\)/\1:/g; s/.$//')
	echo "MAC addr file not exist or not right: generate MAC addr: $MAC_PRE$MAC_POST"
	echo $MAC_PRE$MAC_POST > ${MAC_PATH}
}

__start()
{
	echo ${FIRMWARE}
	# we don't use reset_card
	# rm -f ${CAMERA_OFF}
	
	# Power up 8887 SoC.
	# SoC power on & rescan is required
	#	1. if it is warmboot power off to power on.
	#	2. if battery is removed and inserted again.
	w8887_RESET="$(cat /sys/devices/platform/dw_mmc_sdio.0/support_card)"
	if [ "${w8887_RESET}" == 0 ]; then
		echo "power sdio up because sdio isn't activated"
		echo 1 > /sys/devices/platform/dw_mmc_sdio.0/support_card
	fi

	# when wifi-test.tg exists in sdcard,
	# we gives more params "auto_ds=2 ps_mode=2 hw_test=1"
	# those are needed for wifi certification, TRP/TIS
	WFT="$(cat /sdcard/wifi-test.tg)"
	if [ "${WFT}" == wftest ]; then
		DRV_PARAM="${DRV_PARAM} ${DRV_PARAM_CERTI}"
	fi

	/bin/sleep 0.1

	MOD_CFG80211="$(lsmod |grep cfg80211)"

	# loading driver modules if those are unloaded.
	if [ -z "${MOD_CFG80211}" ]; then

		/sbin/insmod ${MODULE_PATH}/cfg80211.ko ieee80211_regdom=${reg_dom}
		/sbin/insmod ${MODULE_PATH}/mlan.ko
		echo "insmod ${MODULE_PATH}/sd8xxx.ko ${DRV_PARAM} fw_name=${FIRMWARE} reg_alpha2=${reg_dom}"
		/sbin/insmod ${MODULE_PATH}/sd8xxx.ko ${DRV_PARAM} fw_name=${FIRMWARE} reg_alpha2=${reg_dom}
